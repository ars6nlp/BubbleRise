void FillContent(OceanZone zone, ZoneTheme theme, Func<float, float> profile, float height) { int slotCount = Mathf.Clamp(Mathf.RoundToInt(height / 5.5f), 3, 10); for (int i = 0; i < slotCount; i++) { float t = Mathf.Lerp(0.10f, 0.90f, slotCount > 1 ? i / (float)(slotCount - 1) : 0.5f) + UnityEngine.Random.Range(-0.02f, 0.02f); float y = -height * 0.5f + t * height; float passage = Mathf.Min(profile(t) * _field.PlayHalfWidth, _field.PlayHalfWidth); if (passage < 1.1f) continue; var pattern = (Pattern)UnityEngine.Random.Range(0, 5); var points = BuildPattern(pattern, passage); if (points.Count == 0) continue; float offset = PickOffset(points, passage); for (int j = 0; j < points.Count; j++) { var point = points[j]; float x = point.x + offset; float radius = Mathf.Min(point.z, passage * 0.34f); if (Mathf.Abs(x) > passage - radius) continue; zone.AddBlob(new Vector2(x, y + point.y), radius, theme.BlobTint); } } }

List<Vector3> BuildPattern(Pattern pattern, float passage) { var points = new List<Vector3>(); float small = _cfg.zoneBlobMinRadius; float big = _cfg.zoneBlobMaxRadius; float step = Mathf.Min(big * 2.15f, Mathf.Max(0.8f, passage * 0.62f)); switch (pattern) { case Pattern.Pyramid: { int rows = passage > 2.2f ? 3 : 2; float r = Mathf.Lerp(small, big, 0.45f); for (int row = 0; row < rows; row++) { int count = row + 1; for (int c = 0; c < count; c++) { float x = (c - (count - 1) * 0.5f) * step; points.Add(new Vector3(x, -row * step * 0.92f, r)); } } break; } case Pattern.Arc: { int count = passage > 2.4f ? 7 : 5; float radius = Mathf.Min(passage * 0.85f, 2.6f); float r = Mathf.Lerp(small, big, 0.30f); for (int i = 0; i < count; i++) { float a = Mathf.Lerp(165f, 15f, i / (float)(count - 1)) * Mathf.Deg2Rad; points.Add(new Vector3(Mathf.Cos(a) * radius, Mathf.Sin(a) * radius * 0.55f, r)); } break; } case Pattern.Ridge: { int count = 5; int dir = UnityEngine.Random.value < 0.5f ? -1 : 1; for (int i = 0; i < count; i++) { float k = i / (float)(count - 1); float r = Mathf.Lerp(big, small, k); points.Add(new Vector3(dir * (k - 0.5f) * passage * 1.5f, (k - 0.5f) * 2.2f, r)); } break; } case Pattern.Cluster: { int count = 3; for (int i = 0; i < count; i++) { float r = UnityEngine.Random.Range(big * 0.8f, big); points.Add(new Vector3( UnityEngine.Random.Range(-passage * 0.6f, passage * 0.6f), UnityEngine.Random.Range(-1.3f, 1.3f), r)); } break; } default: { float r = Mathf.Lerp(small, big, 0.5f); float side = Mathf.Max(0.9f, passage * 0.62f); for (int i = 0; i < 3; i++) { float y = (i - 1) * r * 2.1f; points.Add(new Vector3(-side, y, r)); points.Add(new Vector3(side, y, r)); } break; } } return points; }

float PickOffset(List<Vector3> points, float passage) { float limit = Mathf.Max(0f, passage * 0.55f); float best = 0f; int bestCount = int.MaxValue; for (int attempt = 0; attempt < 10; attempt++) { float candidate = limit > 0.01f ? UnityEngine.Random.Range(-limit, limit) : 0f; int blocked = 0; for (int i = 0; i < points.Count; i++) if (Mathf.Abs(points[i].x + candidate) < points[i].z + 0.30f) blocked++; if (blocked <= Difficulty.MaxCenterBlocks(_cfg.zoneMaxCenterBlobs)) return candidate; if (blocked < bestCount) { bestCount = blocked; best = candidate; } } return best; }
